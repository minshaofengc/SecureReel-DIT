cmake_minimum_required(VERSION 3.22)
project(securereel_mobile_core LANGUAGES CXX)
add_library(securereel_mobile_core STATIC src/core.cpp src/hashing.cpp src/posix.cpp)
target_compile_features(securereel_mobile_core PUBLIC cxx_std_17)
target_include_directories(securereel_mobile_core PUBLIC include)
if(APPLE)
  target_compile_options(securereel_mobile_core PRIVATE -Wno-deprecated-declarations)
endif()
option(SECUREREEL_HOST_TESTS "Build native host behavior tests" OFF)
if(SECUREREEL_HOST_TESTS)
  enable_testing()
  add_executable(securereel_core_test tests/core_test.cpp)
  target_link_libraries(securereel_core_test PRIVATE securereel_mobile_core)
  add_test(NAME securereel_core_behavior COMMAND securereel_core_test)
endif()
